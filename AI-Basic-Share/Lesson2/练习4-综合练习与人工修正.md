```python
# example_todo_manager.py
from datetime import datetime


class TodoItem:
    def __init__(self, title, priority="medium"):
        self.title = title
        self.priority = priority
        self.completed = False
        self.created_at = datetime.now()
        self.completed_at = None

    def mark_done(self):
        self.completed = True
        self.completed_at = datetime.now()

    def __str__(self):
        status = "✓" if self.completed else "○"
        return f"[{status}] {self.title} ({self.priority})"


class TodoList:
    def __init__(self):
        self.items = []

    def add(self, title, priority="medium"):
        item = TodoItem(title, priority)
        self.items.append(item)
        return item

    def remove(self, index):
        if 0 <= index < len(self.items):
            return self.items.pop(index)
        return None

    def complete(self, index):
        if 0 <= index < len(self.items):
            self.items[index].mark_done()
            return True
        return False

    def list_by_priority(self, priority):
        return [item for item in self.items if item.priority == priority]

    def get_stats(self):
        total = len(self.items)
        done = sum(1 for item in self.items if item.completed)
        pending = total - done
        return {"total": total, "completed": done, "pending": pending}

    def clear_completed(self):
        self.items = [item for item in self.items if not item.completed]
```
