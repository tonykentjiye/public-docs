## 示例代码

```python
# example_calendar.py
from datetime import datetime, timedelta
from typing import Optional


def generate_weekly_calendar(
    base_date: Optional[str] = None,
    week_start: int = 0,
    include_weekend: bool = False,
) -> list[dict]:
    """
    生成一周的日历日期列表。

    Args:
        base_date: 基准日期字符串 (YYYY-MM-DD)，默认使用当天
        week_start: 周起始日，0=周一，6=周日
        include_weekend: 是否包含周末

    Returns:
        包含日期信息的字典列表
    """
    if base_date:
        try:
            current = datetime.strptime(base_date, "%Y-%m-%d")
        except ValueError:
            return []
    else:
        current = datetime.now()

    # 计算本周起始日
    days_from_start = (current.weekday() - week_start) % 7
    week_begin = current - timedelta(days=days_from_start)
    week_begin = week_begin.replace(hour=0, minute=0, second=0, microsecond=0)

    result = []
    for i in range(7):
        day = week_begin + timedelta(days=i)
        is_weekend = day.weekday() >= 5

        if is_weekend and not include_weekend:
            continue

        result.append(
            {
                "date": day.strftime("%Y-%m-%d"),
                "weekday": day.weekday(),
                "is_weekend": is_weekend,
                "label": day.strftime("%A"),
            }
        )

    return result
```
