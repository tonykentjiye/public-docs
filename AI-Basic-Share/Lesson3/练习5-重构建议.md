## 示例代码（可维护性较差）

```python
# example_legacy_order.py
from datetime import datetime


def process_order(order_data):
    # 这个函数太长了，做了太多事情
    # 验证订单
    if not order_data:
        return {"code": 400, "msg": "订单数据不能为空"}
    if "user_id" not in order_data:
        return {"code": 400, "msg": "缺少用户ID"}
    if "items" not in order_data:
        return {"code": 400, "msg": "缺少商品列表"}
    if len(order_data["items"]) == 0:
        return {"code": 400, "msg": "商品列表不能为空"}
    for item in order_data["items"]:
        if "product_id" not in item:
            return {"code": 400, "msg": "商品缺少product_id"}
        if "quantity" not in item:
            return {"code": 400, "msg": "商品缺少quantity"}
        if item["quantity"] <= 0:
            return {"code": 400, "msg": "商品数量必须大于0"}
        if "price" not in item:
            return {"code": 400, "msg": "商品缺少price"}
        if item["price"] < 0:
            return {"code": 400, "msg": "商品价格不能为负数"}

    # 计算金额
    subtotal = 0
    for item in order_data["items"]:
        subtotal = subtotal + item["price"] * item["quantity"]

    # 计算折扣
    discount = 0
    if "coupon_code" in order_data:
        code = order_data["coupon_code"]
        if code == "SAVE10":
            discount = subtotal * 0.1
        elif code == "SAVE20":
            if subtotal >= 100:
                discount = subtotal * 0.2
            else:
                discount = 0
        elif code == "FREESHIP":
            discount = 0  # 免运费，这里暂不处理
        else:
            pass  # 无效优惠券

    # 计算运费
    shipping = 0
    if subtotal < 50:
        shipping = 5.99
    elif subtotal < 100:
        shipping = 3.99
    else:
        shipping = 0

    total = subtotal - discount + shipping

    # 保存订单
    order_id = str(datetime.now().timestamp()).replace(".", "")
    order_record = {
        "order_id": order_id,
        "user_id": order_data["user_id"],
        "items": order_data["items"],
        "subtotal": round(subtotal, 2),
        "discount": round(discount, 2),
        "shipping": round(shipping, 2),
        "total": round(total, 2),
        "status": "pending",
        "created_at": str(datetime.now()),
    }

    # 发送通知
    print(f"[通知] 订单 {order_id} 已创建，总金额: {total}")
    # 实际项目中这里会调用消息队列

    return {"code": 200, "msg": "success", "data": order_record}
```
