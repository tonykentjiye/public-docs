## 练习用代码

将以下代码保存为 `demo_order_service.py` 后使用 Copilot 解释：

```python
# demo_order_service.py
from datetime import datetime
from typing import Optional


class OrderService:
    def __init__(self, db_connection):
        self.db = db_connection
        self.status_flow = {
            "pending": ["confirmed", "cancelled"],
            "confirmed": ["processing", "cancelled"],
            "processing": ["shipped", "cancelled"],
            "shipped": ["delivered", "returning"],
            "delivered": ["returning", "completed"],
            "returning": ["returned"],
            "returned": [],
            "cancelled": [],
            "completed": [],
        }

    def update_status(self, order_id: int, new_status: str) -> dict:
        order = self.db.query(
            "SELECT * FROM orders WHERE id = ?", (order_id,)
        ).fetchone()

        if not order:
            return {"success": False, "error": "订单不存在"}

        current_status = order["status"]

        if new_status not in self.status_flow.get(current_status, []):
            return {
                "success": False,
                "error": f"不允许从 {current_status} 变更为 {new_status}",
            }

        self.db.execute(
            "UPDATE orders SET status = ?, updated_at = ? WHERE id = ?",
            (new_status, datetime.now(), order_id),
        )
        self.db.commit()

        self._notify_status_change(order_id, current_status, new_status)
        return {"success": True, "new_status": new_status}

    def _notify_status_change(self, order_id: int, old: str, new: str) -> None:
        print(f"[通知] 订单 {order_id}: {old} -> {new}")
```
